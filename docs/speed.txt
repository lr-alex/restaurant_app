TERMINAL 2 - PREPARE CUSTOMER ACCESS
====================================

cd D:\study\hkmu\comp_3810_sever-side\proj\restaurant_app

$base = "http://localhost:8099"

$customer = New-Object Microsoft.PowerShell.Commands.WebRequestSession

$customerLogin = Invoke-RestMethod `
  -Method Post `
  -Uri "$base/api/session" `
  -WebSession $customer `
  -ContentType "application/json" `
  -Body (@{
    username = "customer1"
    password = "Customer@123"
  } | ConvertTo-Json -Compress)

$customerLogin | ConvertTo-Json -Depth 5

Expected:

username = customer1
role = customer
profile.name = Bob Johnson


CREATE 1 - CREATE A BOOKING
===========================

$bookingDate = (Get-Date).Date.AddDays(7)

while ($bookingDate.DayOfWeek -eq "Monday") {
  $bookingDate = $bookingDate.AddDays(1)
}

$bookingCreateBody = @{
  branchNo = 1
  bookingDate = $bookingDate.ToString("yyyy-MM-dd")
  slotStart = "19:00"
  slotEnd = "21:00"
  partySize = 2
  notes = "CRUD create 1"
}

$bookingCreate = Invoke-RestMethod `
  -Method Post `
  -Uri "$base/api/bookings" `
  -WebSession $customer `
  -ContentType "application/json" `
  -Body ($bookingCreateBody | ConvertTo-Json -Compress)

$bookingId = $bookingCreate.data._id

$bookingCreate.data | Format-List

Expected:

branchNo = 1
partySize = 2
status = pending

READ 2 - READ STAFF ORDERS
==========================

$staff = New-Object Microsoft.PowerShell.Commands.WebRequestSession

$staffLogin = Invoke-RestMethod `
  -Method Post `
  -Uri "$base/api/session" `
  -WebSession $staff `
  -ContentType "application/json" `
  -Body (@{
    username = "staff1"
    password = "Staff@123"
  } | ConvertTo-Json -Compress)

$staffOrders = (
  Invoke-RestMethod `
    -Method Get `
    -Uri "$base/api/orders" `
    -WebSession $staff
).data

$staffOrders |
  Select-Object _id, branchNo, status, totalAmount |
  Format-Table

$orderId = (
  $staffOrders |
    Where-Object { $_.status -eq "placed" } |
    Select-Object -First 1
)._id

Expected:

The response contains only Branch 1 orders.
One order has status placed and is stored in $orderId.
